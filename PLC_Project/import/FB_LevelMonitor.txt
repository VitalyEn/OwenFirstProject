FUNCTION_BLOCK FB_LevelMonitor
VAR_INPUT
    High, Mid, Low, Flood : BOOL;
END_VAR
VAR_OUTPUT
    State : E_LevelState;
    AllowRun : BOOL;
    NeedFill : BOOL;
    EmergencyStop : BOOL;
END_VAR
IF Flood OR Low THEN
    State := LEV_FAULT;
    AllowRun := FALSE;
    NeedFill := FALSE;
    EmergencyStop := TRUE;
ELSIF High THEN
    State := LEV_HIGH;
    AllowRun := TRUE;
    NeedFill := FALSE;
    EmergencyStop := FALSE;
ELSIF Mid THEN
    State := LEV_MID;
    AllowRun := TRUE;
    NeedFill := FALSE;
    EmergencyStop := FALSE;
ELSE
    State := LEV_LOW;
    AllowRun := FALSE;
    NeedFill := TRUE;
    EmergencyStop := FALSE;
END_IF
END_FUNCTION_BLOCK
